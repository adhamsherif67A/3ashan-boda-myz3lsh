cmake_minimum_required(VERSION 3.8)
project(mako_ps4_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
add_executable(ps4_motion src/ps4_motion.cpp)
target_compile_features(ps4_motion PRIVATE cxx_std_17)
target_compile_options(ps4_motion PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(ps4_motion PRIVATE include)
ament_target_dependencies(ps4_motion rclcpp sensor_msgs mavros_msgs)
install(TARGETS ps4_motion DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_controls test/test_controls.cpp)
  target_compile_features(test_controls PRIVATE cxx_std_17)
  target_include_directories(test_controls PRIVATE include)
  add_test(NAME control_mapping COMMAND test_controls)
endif()
ament_package()
